cmake_minimum_required(VERSION 3.24)
project(TagExplorerNative C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(sqlite3 URL https://www.sqlite.org/2025/sqlite-amalgamation-3500400.zip DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(sqlite3)
add_library(sqlite STATIC ${sqlite3_SOURCE_DIR}/sqlite3.c)
target_include_directories(sqlite PUBLIC ${sqlite3_SOURCE_DIR})
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_DEFAULT_MEMSTATUS=0)
add_executable(TagExplorer WIN32 src/main.c src/catalog.c src/images.c src/icons.c)
target_link_libraries(TagExplorer PRIVATE sqlite ole32 windowscodecs comctl32 comdlg32 shell32 shlwapi gdi32 user32 uuid uxtheme advapi32)
target_compile_definitions(TagExplorer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
 target_compile_options(TagExplorer PRIVATE /W4 /O2 /utf-8)
 target_link_options(TagExplorer PRIVATE /SUBSYSTEM:WINDOWS /OPT:REF /OPT:ICF)
else()
 target_compile_options(TagExplorer PRIVATE -Wall -Wextra -O2)
endif()
